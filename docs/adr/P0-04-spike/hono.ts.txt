import { trpcServer } from "@hono/trpc-server";
import { Hono } from "hono";

import { staffRouter, type SpikeContext } from "./trpc";

type Bindings = { SPIKE_VAR?: string };

export const app = new Hono<{ Bindings: Bindings }>().basePath("/api");

app.get("/health", (c) =>
  c.json({
    ok: true,
    // Hono's own view of env / executionCtx, as passed to app.fetch(request, env, ctx).
    honoEnvVar: c.env.SPIKE_VAR,
    honoHasExecutionCtx: typeof c.executionCtx?.waitUntil === "function",
  }),
);

app.use(
  "/trpc/staff/*",
  trpcServer({
    router: staffRouter,
    endpoint: "/api/trpc/staff",
    createContext: (_opts, c): SpikeContext => ({
      requestId: crypto.randomUUID(),
      spikeVar: c.env.SPIKE_VAR,
      waitUntil: (p) => c.executionCtx.waitUntil(p),
      via: "hono",
    }),
  }),
);
